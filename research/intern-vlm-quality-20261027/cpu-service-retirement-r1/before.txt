# /etc/systemd/system/opi-vlm-model.service
[Unit]
Description=SmolVLM 256M CPU inference
After=network.target

[Service]
User=opi-vlm
Group=opi-vlm
WorkingDirectory=/opt/opi-vlm
ExecStart=/opt/opi-vlm/bin/llama-server --model /opt/opi-vlm/models/SmolVLM-256M-Instruct-Q8_0.gguf --mmproj /opt/opi-vlm/models/mmproj-SmolVLM-256M-Instruct-f16.gguf --alias smolvlm-256m --host 127.0.0.1 --port 8081 --ctx-size 4096 --parallel 1 --threads 2 --threads-batch 2 --n-gpu-layers 0 --no-mmproj-offload --cache-ram 128 --no-context-shift
CPUAffinity=6 7
Restart=on-failure
RestartSec=5
TimeoutStopSec=30
NoNewPrivileges=true
PrivateTmp=true
ProtectHome=true
ProtectSystem=strict
MemoryMax=4G

[Install]
WantedBy=multi-user.target
